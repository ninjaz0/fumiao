cmake_minimum_required(VERSION 3.20)
project(fumiao_image_decode LANGUAGES CXX)
find_path(HEIF_INCLUDE libheif/heif.h REQUIRED)
find_library(HEIF_LIBRARY heif REQUIRED)
add_executable(heif-decode heif-decode.cc)
target_compile_features(heif-decode PRIVATE cxx_std_20)
target_include_directories(heif-decode PRIVATE "${HEIF_INCLUDE}")
target_link_libraries(heif-decode PRIVATE "${HEIF_LIBRARY}")
if(APPLE)
  set_target_properties(heif-decode PROPERTIES INSTALL_RPATH "@executable_path/../lib")
else()
  set_target_properties(heif-decode PROPERTIES INSTALL_RPATH "$ORIGIN/../lib")
endif()
install(TARGETS heif-decode RUNTIME DESTINATION bin)
